# 02 špikeris

3. tēmas špikeris - JavaScript pamati

Mainīgie, datu tipi, sazarojumi, masīvi un cikli. Katrs bloks atbilst vienai stundai.

3.1 Mainīgie un funkcijas

// script.js - pieslēgts ar <script src="script.js"></script> pirms </body>

let punkti = 40;                          // vērtību var mainīt
let dzivibas = 3;
const SPELES_NOSAUKUMS = "Klases cīņa";   // const vairs mainīt NEDRĪKST

function atjaunotEkranu() {
    document.getElementById("punkti").textContent = punkti;
}

atjaunotEkranu();     // funkcija izpildās TIKAI kad to izsauc

// let    parasts mainīgais
// const  nemainīga vērtība (const punkti = 0; punkti = 5; -> TypeError)
// Katras komandas beigās semikols.
// JavaScript atšķir lielos un mazos burtus: Punkti != punkti

Spēles datus definē ārpus funkcijām - citādi tie pazūd, tiklīdz funkcija beidzas.

3.2 Datu tipi un ievade

let punkti = 0;

function pievienotPunktus() {
    let ievade = prompt("Cik punktu pievienot?");  // prompt VIENMĒR atdod tekstu
    let jauni = Number(ievade);                    // teksts "10" -> skaitlis 10

    if (isNaN(jauni)) {                            // vai tiešām sanāca skaitlis?
        alert("Lūdzu ievadi skaitli!");
        return;
    }

    punkti = punkti + jauni;
    document.getElementById("punkti").textContent = punkti;
}

// BEZ Number():   "10" + 5  ->  "105"   (teksti tiek salikti kopā)
// AR Number():     10  + 5  ->   15

// Datu tipi: string "teksts" , number 42 , boolean true/false
typeof punkti      // "number"

Ja skaitļi sāk "salipt kopā" (105 vietā 15), tu esi aizmirsis Number().

3.3 Sazarojumi un loģiskie operatori

let punkti = 150;
let dzivibas = 0;

function parbauditStavokli() {
    let statuss = document.getElementById("statuss");

    // SVARĪGI: vissvarīgākais nosacījums vienmēr ir PIRMAIS
    if (dzivibas <= 0) {
        statuss.textContent = "Spēle beigusies!";
    } else if (punkti >= 100) {
        statuss.textContent = "Lielisks rezultāts!";
    } else {
        statuss.textContent = "Turpini spēlēt";
    }
}

// ===  vienāds (arī tips)      !==  nav vienāds
// &&   UN (abiem jābūt)       ||   VAI (pietiek ar vienu)
// !    NE (apgriež otrādi)

if (punkti >= 100 && dzivibas > 0) { /* abiem jābūt patiesiem */ }

JavaScript lieto ===, nevis ==. Ar vienu = tu piešķir vērtību, nevis salīdzini.

3.4 Masīvi

let inventars = ["marķieris", "plāns", "lapiņa"];

inventars.length          // 3           cik elementu
inventars[0]              // "marķieris" pirmais ir 0, nevis 1
inventars[inventars.length - 1]          // pēdējais
inventars.push("zīmulis") //             pievieno beigās
inventars.pop()           //             izņem pēdējo
inventars.includes("plāns")              // true / false

function paraditInventaru() {
    let vieta = document.getElementById("inventars-parads");

    // Bez šī nosacījuma tukša soma lapā izskatītos kā pazudusi
    if (inventars.length === 0) {
        vieta.textContent = "Soma ir tukša";
        return;
    }
    vieta.textContent = inventars.join(", ");
}

inventars[3] trīs elementu masīvā dod undefined - nevis kļūdu. Tāpēc tā ir grūti pamanāma.

3.5-3.6 Cikli un teksta spēle

let inventars = ["marķieris", "plāns", "lapiņa"];

function zimetSarakstu() {
    let saraksts = document.getElementById("inventara-saraksts");

    // Bez šīs rindas katrs izsaukums pieliktu sarakstu vēlreiz klāt
    saraksts.innerHTML = "";

    for (let i = 0; i < inventars.length; i++) {
        saraksts.innerHTML += "<li>" + inventars[i] + "</li>";
    }
}

// for cikls: sākums ; nosacījums ; solis
for (let i = 0; i < 5; i++) { }     // i = 0,1,2,3,4

// Pār visiem masīva elementiem
for (let lieta of inventars) { console.log(lieta); }

// while - kamēr nosacījums ir patiess
let dzivibas = 3;
while (dzivibas > 0) { dzivibas--; }   // bez dzivibas-- lapa iesalst!

Pirms zīmēt sarakstu no jauna, vienmēr notīri veco ar innerHTML = "".
