9.2 API izsaukums ar requests

import requests

def laiks_riga():
    try:
        r = requests.get(
            "https://api.open-meteo.com/v1/forecast",
            params={"latitude": 56.95, "longitude": 24.10,
                    "current_weather": True},
            timeout=5                      # bez šī programma var karāties mūžīgi
        )
        if r.status_code != 200:           # 200 = viss kārtībā
            return None
        return r.json()["current_weather"]
    except requests.exceptions.RequestException:
        print("Nav savienojuma ar serveri")
        return None

dati = laiks_riga()
if dati:
    print(dati["temperature"])

# Statusa kodi: 200 OK , 404 nav atrasts , 500 servera kļūda

Vienmēr pārbaudi status_code un liec timeout - internets var arī nestrādāt.

9.4-9.5 HTML/CSS un session_state

import streamlit as st

# Savs HTML - vajag unsafe_allow_html=True, citādi redzēsi tagus kā tekstu
st.markdown("<h1 style='color:#FFB142;'>Virsraksts</h1>",
            unsafe_allow_html=True)

col1, col2 = st.columns([2, 1])       # pirmā kolonna divreiz platāka
with col1:
    st.write("Kreisā puse")

# session_state - BEZ ŠĪ lapa aizmirst visu pie katra klikšķa
if "klikski" not in st.session_state:      # inicializē tikai pirmoreiz
    st.session_state.klikski = 0

if st.button("Klikšķini"):
    st.session_state.klikski += 1
st.write(f"Klikšķi: {st.session_state.klikski}")

# Forma - iztukšojas pēc nosūtīšanas
with st.form("forma", clear_on_submit=True):
    jauns = st.text_input("Jauns uzdevums")
    ok = st.form_submit_button("Pievienot")

# Katrai pogai ciklā vajag SAVU key
for i, u in enumerate(st.session_state.todo):
    st.button("Dzēst", key=f"del_{i}")
