Procesa analīze (1. uzdevums, 1.–3. solis) Tabulas nav vai tajā ir mazāk par trim soļiem. Soļi ir, bet A/D/N vai Plānotājā? nav pamatots. Vismaz pieci soļi, katram A/D/N un pamatojums gan tam, kas automatizēts, gan tam, kas nav. Pamatots arī vismaz viens A solis, ko apzināti neautomatizēja (piemēram, atļauju vai drošības dēļ).
Skripts (2. uzdevums, 4.–8. solis) Izvēlne neparādās, vai skripts beidzas ar kļūdu. Strādā viena komanda, vai atkārtota palaišana dubulto sarakstu. Abas komandas pareizi strādā ar jaunajām rindām un statusa maiņu, nekas nedubultojas, un kods ir planotajs_76.js. Pievienots papildu uzdevums vai cits paša uzlabojums, kas aprakstīts README.md.
Piekļuves tiesības (3. uzdevums, 1.–4. solis) Koplietots tikai ar saiti vai visiem ar vienu lomu. Lomas atbilst prasībām, bet nav pārbaudes tabulas. Trīs lomas pēc prasībām, un pārbaudes tabulā visas rindas sakrīt. README.md paskaidro, ko skolas saite dod skolotājam, un kā to novērst.
Lietošanas instrukcija (3. uzdevums, 5.–7. solis) Nav PDF, vai teksts ir bez stiliem. Stili ir, bet satura rādītājs vai attēlu numuri rakstīti ar roku. PDF ar automātisku satura rādītāju un trim ilustrācijām ar Word parakstiem. Ir arī mijnorāde uz ilustrāciju vai galvene ar lappušu numuriem.

const LAPA = 'Uzdevumi';   // kolonnas: A uzdevums, B atbildīgais, C termiņš, D statuss

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Plānotājs')
    .addItem('Atzīmēt nokavētos', 'atzimetNokavetos')
    .addItem('Saraksts pa cilvēkiem', 'sarakstsPaCilvekiem')
    .addToUi();
}

function atzimetNokavetos() {
  const lapa = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(LAPA);
  const dati = lapa.getDataRange().getValues();
  const sodien = new Date();
  sodien.setHours(0, 0, 0, 0);
  let nokaveti = 0;
  for (let i = 1; i < dati.length; i++) {
    if (dati[i][0] === '') {
      continue;
    }
    const termins = dati[i][2];
    const statuss = dati[i][3];
    const rinda = lapa.getRange(i + 1, 1, 1, 4);   // tabulas rindas sākas ar 1
    if (!(termins instanceof Date)) {
      rinda.setBackground('#fff2cc');              // dzeltens: pārbaudi datumu
    } else if (termins < sodien && statuss !== 'Gatavs') {
      rinda.setBackground('#f4cccc');              // sarkans: nokavēts
      nokaveti += 1;
    } else {
      rinda.setBackground(null);                   // bez krāsas
    }
  }
  SpreadsheetApp.getUi().alert('Nokavēti uzdevumi: ' + nokaveti);
}

function sarakstsPaCilvekiem() {
  const tabula = SpreadsheetApp.getActiveSpreadsheet();
  const dati = tabula.getSheetByName(LAPA).getDataRange().getValues();
  let pec = {};
  for (let i = 1; i < dati.length; i++) {
    const uzdevums = dati[i][0];
    const kurs = dati[i][1];
    if (uzdevums === '' || dati[i][3] === 'Gatavs') {
      continue;
    }
    if (pec[kurs] === undefined) {
      pec[kurs] = [];
    }
    pec[kurs].push(uzdevums);
  }
  let lapa = tabula.getSheetByName('Pa cilvēkiem');
  if (lapa === null) {
    lapa = tabula.insertSheet('Pa cilvēkiem');
  }
  lapa.clear();
  lapa.appendRow(['Atbildīgais', 'Uzdevumi']);
  for (let kurs in pec) {
    lapa.appendRow([kurs, pec[kurs].join(', ')]);
  }
}
