4.3 Stāvokļu mašīna: viens stāvoklis vienlaikus

public enum EnemyState { Idle, Patrol, Chase, Attack, Dead }

public partial class Enemy : CharacterBody2D
{
    private EnemyState state = EnemyState.Idle;
    private double stateTime = 0.0;
    private Player player;

    private void ChangeState(EnemyState nextState)
    {
        state = nextState;
        stateTime = 0.0;
    }

    public override void _PhysicsProcess(double delta)
    {
        stateTime += delta;
        double distance = Position.DistanceTo(player.Position);

        // Slieksnis ATPAKAĻ ir LIELĀKS nekā uz priekšu -> nemirgo uz robežas
        if (state != EnemyState.Dead)
        {
            if (distance < 50) ChangeState(EnemyState.Attack);
            else if (distance < 300) ChangeState(EnemyState.Chase);
            else if (distance > 400) ChangeState(EnemyState.Patrol);
        }

        switch (state)
        {
            case EnemyState.Patrol: Patrol(); break;
            case EnemyState.Chase: Chase(); break;
            case EnemyState.Attack: Attack(); break;
            case EnemyState.Dead: break;   // gala stāvoklis
        }
    }
}

DEAD ir gala stāvoklis - no tā nav nevienas izejošas pārejas.

private void Chase()
{
    NavigationAgent2D agent = GetNode<NavigationAgent2D>("NavigationAgent2D");
    agent.TargetPosition = player.Position;

    Vector2 next = agent.GetNextPathPosition();
    Vector2 dir = (next - Position).Normalized();

    Velocity = dir * speed;
    MoveAndSlide();
}

4.5 Object pooling: neveido un nedzēs, bet lieto atkārtoti

public class BulletPool
{
    private readonly List<Bullet> pool = new();
    private const int PoolSize = 50;

    public void Initialize(Node parent)
    {
        for (int i = 0; i < PoolSize; i++)
        {
            Bullet bullet = new Bullet();
            bullet.Visible = false;
            bullet.SetProcess(false);
            parent.AddChild(bullet);
            pool.Add(bullet);
        }
    }

    public Bullet GetBullet()
    {
        foreach (Bullet bullet in pool)
            if (!bullet.Visible) return bullet;    // atkārtoti lieto neaktīvo
        return null;                               // visi aizņemti
    }
}

// Bez pooling: 200 šāvieni = 200 Instantiate + 200 QueueFree -> FPS krīt
// Ar pooling:  mezglu skaits scēnā paliek nemainīgs

Ja GetBullet() atgriež null, pārbaudi to - citādi programma avarēs.
