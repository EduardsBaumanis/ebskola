Teorija: 2D orientācija plaknē un daudzkomponentu sistēmas arhitektūra

2D virziena vektora normalizācija un orientācija pret kursoru (LookAt)

Apvienojot horizontālo un vertikālo ievadi 2D plaknē, iegūst astoņu virzienu kustības vektoru. Virzoties pa diagonāli ar vektoru (1, 1), tā garums pēc Pitagora teorēmas ir √(1² + 1²) ≈ 1,41, tāpēc bez korekcijas kustība slīpi būtu par aptuveni 41% ātrāka nekā taisni pa asi. Metode .Normalized() (uz nenulles vektora) samazina vektora garumu līdz precīzam vieninieka garumam 1, saglabājot nemainīgu kustības ātrumu visos virzienos.

Lai Node2D mezglu pagrieztu pret konkrētu punktu pasaules koordinātās, kursoru nolasa ar GetGlobalMousePosition() un izsauc LookAt(merkaPunkts): mezgls pagriežas tā, lai tā lokālā priekšējā ass +X (Transform.X) rādītu tieši uz mērķi:

// 1. Astoņu virzienu ievade un normalizācija vienādam ātrumam:
Vector2 ievade = new Vector2(
    Input.GetAxis("move_left", "move_right"),
    Input.GetAxis("move_up", "move_down")
);
if (ievade != Vector2.Zero)
    ievade = ievade.Normalized(); // Diagonāles garums 1.41 -> 1.0
Velocity = ievade * Atrums;
MoveAndSlide();

// 2. Pagriešana pret kursoru un priekšējā virziena nolasīšana:
Vector2 merkaPunkts = GetGlobalMousePosition();
LookAt(merkaPunkts);
Vector2 prieksejaisVirziens = Transform.X; // Vienības vektors (+X ass)

Modulāra arhitektūra, polimorfisms un posmu pārvaldība

Apvienojot vairākas apakšsistēmas vienā reāllaika lietotnē, tīra arhitektūra (architecture) nošķir katras klases pienākumus — līdzīgi kā lidostas dispečeru tornis plāno reisu secību un atjauno informācijas tablo, bet katra tehniskā vienība uz skrejceļa pati apbrauc šķēršļus pēc savas lomas noteikumiem:

  • Mantošana un polimorfisms: kopīgo stāvokļu mašīnu un ceļa meklēšanu ievieto bāzes klasē, bet specializētos kustības vai darbības parametrus — atvasinātajās klasēs. Glabājot visus aktīvos objektus vienā List<BazesTips> sarakstā, tos apstrādā vienots cikls bez if pārbaudēm katram paveidam.
  • Posmu pārvaldība un datu struktūras: atsevišķs vadītāja mezgls sarakstā (piemēram, izmantojot kompaktu C# datu ierakstu record vai struct) glabā nākamo posmu parametrus un pāriet pie nākamā posma brīdī, kad aktīvo objektu skaits sasniedz nulli (Count == 0), vienlaikus atjaunojot lietotāja saskarnes (UI) rādītājus.
  • Veiktspēja un slāņu izolācija: pieaugot aktīvo objektu skaitam un skaitļošanas sarežģītībai (computational complexity), objektu krātuve (Object Pooling) novērš atmiņas svārstības, savukārt sadursmju slāņi (Collision Layer / Mask) nodrošina, ka objekta radītās saskares zonas nereaģē pašas ar savu avotu:
// Kompakts datu ieraksts (record) posma konfigurācijai:

public partial class Dispecers : Node

using Godot;

public partial class Player : CharacterBody2D
{
    [Export] public float Speed { get; set; } = 240.0f;
    [Export] public int MaxHp { get; set; } = 100;
    [Export] public PackedScene BulletScene { get; set; } = GD.Load<PackedScene>("res://Scenes/Bullet.tscn");
    [Export] public bool UsePooling { get; set; } = true;
    [Export] public int PoolSize { get; set; } = 50;
    [Export] public float FireCooldown { get; set; } = 0.18f;

    public int Hp { get; private set; } = 100;
    public bool IsAlive => Hp > 0;

    private readonly BulletPool bulletPool = new();
    private double cooldownTimer = 0.0;

    public override void _Ready()
    {
        Hp = MaxHp;
        CallDeferred(nameof(SetupPool));
    }

    private void SetupPool()
    {
        bulletPool.Initialize(GetParent(), BulletScene, PoolSize);
    }

    public override void _PhysicsProcess(double delta)
    {
        if (!IsAlive)
        {
            Velocity = Vector2.Zero;
            return;
        }

        Vector2 inputDir = new Vector2(
            Input.GetAxis("move_left", "move_right"),
            Input.GetAxis("move_up", "move_down")
        );

        if (inputDir != Vector2.Zero)
        {
            inputDir = inputDir.Normalized();
        }

        Velocity = inputDir * Speed;
        MoveAndSlide();

        LookAt(GetGlobalMousePosition());

        cooldownTimer -= delta;
        if (Input.IsActionPressed("shoot") && cooldownTimer <= 0.0)
        {
            Shoot();
            cooldownTimer = FireCooldown;
        }
    }

    private void Shoot()
    {
        if (BulletScene == null) return;

        if (UsePooling)
        {
            Bullet pooledBullet = bulletPool.GetBullet();
            if (pooledBullet == null) return;

            pooledBullet.Fire(GlobalPosition, Transform.X, true);
        }
        else
        {
            Bullet newBullet = BulletScene.Instantiate<Bullet>();
            GetParent().AddChild(newBullet);
            newBullet.Fire(GlobalPosition, Transform.X, false);
        }
    }

    public void TakeDamage(int amount)
    {
        if (!IsAlive || amount <= 0) return;

        Hp = Mathf.Max(0, Hp - amount);
        GD.Print($"[Player] -{amount} HP (atlicis {Hp}/{MaxHp})");

        if (Hp == 0)
        {
            Modulate = new Color(0.4f, 0.4f, 0.4f, 0.7f);
            GD.Print("[Player] Spēlētājs ir uzveikts!");
        }
    }

    public void Heal(int amount)
    {
        if (!IsAlive || amount <= 0) return;
        Hp = Mathf.Min(MaxHp, Hp + amount);
    }
}

    private void EnsureComponents()
    {
        if (GetNodeOrNull<CollisionShape2D>("CollisionShape2D") == null)
        {
            CollisionShape2D col = new() { Name = "CollisionShape2D" };
            col.Shape = new CircleShape2D { Radius = 16.0f };
            AddChild(col);
        }

        Polygon2D visual = GetNodeOrNull<Polygon2D>("Visual");
        if (visual == null)
        {
            visual = new Polygon2D
            {
                Name = "Visual",
                Polygon = new Vector2[]
                {
                    new(-16, -16), new(16, -12), new(20, 0), new(16, 12), new(-16, 16)
                }
            };
            AddChild(visual);
        }
        visual.Color = BodyColor;

        if (GetNodeOrNull<NavigationAgent2D>("NavigationAgent2D") == null)
        {
            NavigationAgent2D agent = new()
            {
                Name = "NavigationAgent2D",
                PathDesiredDistance = 16.0f,
                TargetDesiredDistance = 20.0f
            };
            AddChild(agent);
        }
    }

    public void ChangeState(EnemyState newState)
    {
        if (CurrentState == EnemyState.Dead || CurrentState == newState) return;
        CurrentState = newState;
    }

    public override void _PhysicsProcess(double delta)
    {
        if (CurrentState == EnemyState.Dead) return;

        if (player != null && player.IsAlive)
        {
            float dist = GlobalPosition.DistanceTo(player.GlobalPosition);
            if (dist < AttackRange)
            {
                ChangeState(EnemyState.Attack);
            }
            else if (dist < DetectRange)
            {
                ChangeState(EnemyState.Chase);
            }
            else
            {
                ChangeState(EnemyState.Patrol);
            }
        }
        else
        {
            ChangeState(EnemyState.Patrol);
        }

        switch (CurrentState)
        {
            case EnemyState.Idle:
                Velocity = Vector2.Zero;
                break;

            case EnemyState.Patrol:
                Velocity = new Vector2(patrolDirection * (Speed * 0.5f), 0);
                if (Mathf.Abs(GlobalPosition.X - patrolOrigin.X) > 90.0f)
                {
                    patrolDirection *= -1;
                }
                break;

            case EnemyState.Chase:
                ChasePlayer();
                break;

            case EnemyState.Attack:
                Velocity = Vector2.Zero;
                PerformAttack(delta);
                break;
        }

        MoveAndSlide();
    }

    protected virtual void ChasePlayer()
    {
        if (player == null) return;

        if (UsePathfinding && navAgent != null)
        {
            navAgent.TargetPosition = player.GlobalPosition;
            Vector2 nextPoint = navAgent.GetNextPathPosition();
            Vector2 dir = (nextPoint - GlobalPosition).Normalized();
            Velocity = dir * Speed;
            LookAt(nextPoint);
        }
        else
        {
            Vector2 dir = (player.GlobalPosition - GlobalPosition).Normalized();
            Velocity = dir * Speed;
            LookAt(player.GlobalPosition);
        }
    }

    protected virtual void PerformAttack(double delta)
    {
        if (player == null || !player.IsAlive) return;

        LookAt(player.GlobalPosition);
        attackCooldown -= delta;

        if (attackCooldown <= 0.0)
        {
            attackCooldown = AttackInterval;
            player.TakeDamage(AttackDamage);
        }
    }

    public void TakeDamage(int amount)
    {
        if (CurrentState == EnemyState.Dead || amount <= 0) return;

        Hp -= amount;
        if (Hp <= 0)
        {
            Hp = 0;
            Velocity = Vector2.Zero;
            ChangeState(EnemyState.Dead);
        }
    }
}
