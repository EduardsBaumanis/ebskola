2.1 delta - kāpēc bez tā spēle uz cita datora ir citāda

public override void _PhysicsProcess(double delta)
{
    // PAREIZI: ātrums pikseļos SEKUNDĒ
    Position += velocity * (float)delta;

    // NEPAREIZI: ātrums pikseļos KADRĀ
    // Position += velocity;
    //   -> uz 144 Hz datora objekts kustas 2.4x ātrāk nekā uz 60 Hz
}

// Kuru metodi lietot?
public override void _Process(double delta)         // katrā zīmētajā kadrā, delta SVĀRSTĀS
                                                    // -> vizuāli efekti, UI
public override void _PhysicsProcess(double delta)  // fiksēti 60 reizes/s, delta STABILS
                                                    // -> kustība, fizika, sadursmes

Īkšķa likums: ja objekts kustas vai saduras - _PhysicsProcess. Viss pārējais - _Process.

2.2 Ievade un normalizēta kustība

public override void _PhysicsProcess(double delta)
{
    Vector2 dir = Vector2.Zero;

    if (Input.IsActionPressed("move_right")) dir.X += 1;
    if (Input.IsActionPressed("move_left")) dir.X -= 1;
    if (Input.IsActionPressed("move_down")) dir.Y += 1;
    if (Input.IsActionPressed("move_up")) dir.Y -= 1;

    dir = dir.Normalized();     // BEZ ŠĪ diagonāle ir 1.41x ātrāka!
    Velocity = dir * speed;
    MoveAndSlide();
}

// IsActionPressed()      - kamēr TUR taustiņu   -> kustība
// IsActionJustPressed()  - vienu reizi nospiežot -> lēciens, šāviens

Darbību nosaukumus (move_right) veido Project Settings -> Input Map, nevis raksti taustiņu kodus.

2.5 Gravitācija, lēciens, coyote time

private const float Gravity = 1200.0f;
private const float JumpForce = -550.0f;   // NEGATĪVS, jo Y iet uz leju
private double coyote = 0.0;

public override void _PhysicsProcess(double delta)
{
    Vector2 velocity = Velocity;

    if (!IsOnFloor()) velocity.Y += Gravity * (float)delta;

    // coyote time: īsa tolerance pēc nokāpšanas no malas
    if (IsOnFloor()) coyote = 0.1;
    else coyote -= delta;

    if (Input.IsActionJustPressed("jump") && coyote > 0.0)
    {
        velocity.Y = JumpForce;
        coyote = 0.0;
    }

    // mainīgs lēciena augstums: atlaid ātrāk = zemāks lēciens
    if (Input.IsActionJustReleased("jump") && velocity.Y < 0)
        velocity.Y *= 0.5f;

    Velocity = velocity;
    MoveAndSlide();
}

Bez IsOnFloor() vai coyote pārbaudes varonis var lēkt bezgalīgi gaisā.
