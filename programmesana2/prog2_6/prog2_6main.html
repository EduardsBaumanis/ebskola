// Control mezgli NAV Node2D - tiem ir enkuri, konteineri un tēma.
//   Label            teksts            Button       poga
//   LineEdit         viena rinda       TextEdit     vairākas rindas
//   VBoxContainer    kārto vertikāli   HBoxContainer  horizontāli
//   GridContainer    režģis            PanelContainer fons un rāmis
// Konteiners kārto bērnus PATS - nelieto rokas pozīcijas tajā.

void MainMenu::_bind_methods() {
    ClassDB::bind_method(D_METHOD("on_start_pressed"),
                         &MainMenu::on_start_pressed);
}
void MainMenu::_ready() {
    Button *b = get_node<Button>("VBoxContainer/StartButton");
    b->connect("pressed", Callable(this, "on_start_pressed"));
}
void MainMenu::on_start_pressed() {
    get_tree()->change_scene_to_file("res://level1.tscn");
}

6.2 Tween - animācija ar kodu

#include <godot_cpp/classes/tween.hpp>

Ref<Tween> tween = create_tween();

// no pašreizējās vērtības uz (600, 300) vienā sekundē
tween->tween_property(node, "position", Vector2(600, 300), 1.0);

// izgaismošanās
node->set_modulate(Color(1, 1, 1, 0));
tween->tween_property(node, "modulate:a", 1.0, 0.5);

tween->set_parallel(true);                  // visas reizē, nevis pēc kārtas
tween->set_trans(Tween::TRANS_ELASTIC);     // atlec mērķī

// Ja notikums var atkārtoties ātri - apturi veco:
if (active_tween.is_valid()) active_tween->kill();

// AnimationPlayer = vizuāli veidota animācija ar atslēgkadriem
//   player->play("uzvara");

Animācija nav tikai skaistums - tā pasaka spēlētājam, ka darbība tika pamanīta.

6.3 Audio: efekti, mūzika un bus

#include <godot_cpp/classes/audio_stream_player.hpp>

AudioStreamPlayer *sfx = get_node<AudioStreamPlayer>("SFX");
sfx->play();

// Mūzikas mīksta ienākšana
AudioStreamPlayer *music = get_node<AudioStreamPlayer>("Music");
music->set_volume_db(-40);
music->play();
create_tween()->tween_property(music, "volume_db", 0.0, 1.0);

// Atsevišķi bus (apakšā cilne Audio): Master -> Music , SFX
AudioServer *srv = AudioServer::get_singleton();
int idx = srv->get_bus_index("Music");
srv->set_bus_volume_db(idx, -10.0);

// Skaļumu mēra decibelos:  0 dB = pilns , -80 dB = kluss

Mūziku un efektus liec atsevišķos bus - tad spēlētājs var izslēgt mūziku, saglabājot skaņas.
