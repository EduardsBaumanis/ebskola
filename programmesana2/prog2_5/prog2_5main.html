5.1 Failu lasīšana un rakstīšana

#include <godot_cpp/classes/file_access.hpp>

// res://  - projekta faili, TIKAI lasīšanai gatavā spēlē
// user:// - spēlētāja dati, te saglabā (Project -> Open User Data Folder)

Ref<FileAccess> f = FileAccess::open("user://save.txt", FileAccess::WRITE);
if (f.is_valid()) {                       // VIENMĒR pārbaudi
    f->store_string("Sveiks!");
    f->store_32(150);
}

// Nolasīšana ar trūkstoša faila apstrādi (= Python FileNotFoundError)
int load_score() {
    if (!FileAccess::file_exists("user://save.txt")) return 0;
    Ref<FileAccess> f = FileAccess::open("user://save.txt", FileAccess::READ);
    return f->get_32();
}

Nolasīšanas secībai jābūt TIEŠI tādai pašai kā rakstīšanas secībai, citādi dati sajūk.

5.3 Save sistēma ar slotiem un versiju

class Saveable {                       // interfeiss: ko prot katrs objekts
public:
    virtual Dictionary save_data() = 0;
    virtual void load_data(Dictionary d) = 0;
};

class SaveManager {
    std::vector<Saveable*> objects;
public:
    String slot_path(int slot) { return "user://slot" + itos(slot) + ".json"; }

    Dictionary collect() {
        Dictionary all;
        all["version"] = 1;            // versija: vecs fails nesabojā spēli
        for (int i = 0; i < objects.size(); i++)
            all[i] = objects[i]->save_data();
        return all;
    }
};

// Ielādējot:
if ((int)d["version"] != 1) { /* brīdini un lieto noklusējumus */ }

SaveManager nezina, kas ir spēlētāja iekšienē - katrs objekts saglabā sevi pats.

5.4 Procedurālā ģenerēšana ar Random Walk

// 1 = siena, 0 = eja
std::vector<std::vector<int>> grid(15, std::vector<int>(15, 1));

int x = 7, y = 7;                       // sāc no centra
grid[y][x] = 0;

for (int i = 0; i < 200; i++) {
    int dir = UtilityFunctions::randi_range(0, 3);
    if (dir == 0 && x > 1)      x--;     // robežu pārbaude OBLIGĀTA
    else if (dir == 1 && x < 13) x++;
    else if (dir == 2 && y > 1)  y--;
    else if (dir == 3 && y < 13) y++;
    grid[y][x] = 0;
}

// OBLIGĀTI: pārbaudi, vai izeja ir sasniedzama, un ģenerē atkārtoti, ja nav.
// Režģis -> ekrāns:
//   siena.set_position(Vector2(x * 64, y * 64));

Bez izejamības pārbaudes spēlētājs var sākt līmenī, kuru nav iespējams izspēlēt.
