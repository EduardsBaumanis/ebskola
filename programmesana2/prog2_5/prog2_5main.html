5.1 Failu lasīšana un rakstīšana

using Godot;

// res://  - projekta faili, TIKAI lasīšanai gatavā spēlē
// user:// - spēlētāja dati, te saglabā (Project -> Open User Data Folder)

using FileAccess file = FileAccess.Open("user://save.txt", FileAccess.ModeFlags.Write);
if (file != null)                         // VIENMĒR pārbaudi
{
    file.StoreString("Sveiks!");
    file.Store32(150);
}

// Nolasīšana ar trūkstoša faila apstrādi (= Python FileNotFoundError)
int LoadScore()
{
    if (!FileAccess.FileExists("user://save.txt")) return 0;
    using FileAccess readFile = FileAccess.Open("user://save.txt", FileAccess.ModeFlags.Read);
    return readFile != null ? (int)readFile.Get32() : 0;
}

Nolasīšanas secībai jābūt TIEŠI tādai pašai kā rakstīšanas secībai, citādi dati sajūk.

5.3 Save sistēma ar slotiem un versiju

using Godot.Collections;
using System.Collections.Generic;

public interface ISaveable             // interfeiss: ko prot katrs objekts
{
    Dictionary SaveState();
    void LoadState(Dictionary data);
}

public class SaveManager
{
    private readonly List<ISaveable> objects = new();

    public string SlotPath(int slot) => $"user://slot{slot}.json";

    public Dictionary Collect()
    {
        Dictionary all = new();
        all["version"] = 1;          // versija: vecs fails nesabojā spēli
        for (int i = 0; i < objects.Count; i++)
            all[i] = objects[i].SaveState();
        return all;
    }
}

// Ielādējot:
if (data["version"].AsInt32() != 1) { /* brīdini un lieto noklusējumus */ }

SaveManager nezina, kas ir spēlētāja iekšienē - katrs objekts saglabā sevi pats.

5.4 Procedurālā ģenerēšana ar Random Walk

using System;

// 1 = siena, 0 = eja
int[,] grid = new int[15, 15];
for (int yy = 0; yy < 15; yy++)
    for (int xx = 0; xx < 15; xx++)
        grid[yy, xx] = 1;

int x = 7, y = 7;                       // sāc no centra
grid[y, x] = 0;
Random rng = new Random();

for (int i = 0; i < 200; i++)
{
    int dir = rng.Next(0, 4);
    if (dir == 0 && x > 1) x--;          // robežu pārbaude OBLIGĀTA
    else if (dir == 1 && x < 13) x++;
    else if (dir == 2 && y > 1) y--;
    else if (dir == 3 && y < 13) y++;
    grid[y, x] = 0;
}

// OBLIGĀTI: pārbaudi, vai izeja ir sasniedzama, un ģenerē atkārtoti, ja nav.
// Režģis -> ekrāns:
//   wall.Position = new Vector2(x * 64, y * 64);

Bez izejamības pārbaudes spēlētājs var sākt līmenī, kuru nav iespējams izspēlēt.
